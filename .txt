companies
id UUID PRIMARY KEY
name VARCHAR
email VARCHAR UNIQUE
password_hash TEXT
company_name VARCHAR
company_email VARCHAR
company_cert TEXT
access_code TEXT
created_at TIMESTAMP

vehicles
id UUID PRIMARY KEY
company_id UUID FOREIGN KEY REFERENCES companies(id)
type VARCHAR
capacity INT
features TEXT

routes
id UUID PRIMARY KEY
company_id UUID FOREIGN KEY REFERENCES companies(id)
origin VARCHAR
destination VARCHAR
dep_time TIMESTAMP
price FLOAT
terminal JSONB
vehicles JSONB

trips
id UUID PRIMARY KEY
route_id UUID FOREIGN KEY REFERENCES routes(id)
departure_time TIMESTAMP
arrival_time TIMESTAMP
seats_total INT
seats_available INT
price FLOAT

bookings
id UUID PRIMARY KEY
trip_id UUID FOREIGN KEY REFERENCES trips(route_id)
user_id UUID FOREIGN KEY REFERENCES users(id)
seat_number INT
status VARCHAR
created_at TIMESTAMP


payments_user
id UUID PRIMARY KEY
user_id UUID FOREIGN KEY REFERENCES users(id)
booking_id UUID FOREIGN KEY REFERENCES bookings(id)
amount FLOAT
status VARCHAR
reference VARCHAR UNIQUE
created_at TIMESTAMP

payments_company
id UUID PRIMARY KEY
company_id UUID FOREIGN KEY REFERENCES companies(id)
booking_id UUID FOREIGN KEY REFERENCES bookings(id)
amount FLOAT
status VARCHAR
reference VARCHAR UNIQUE
created_at TIMESTAMP
